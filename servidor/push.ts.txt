import { createClient } from "npm:@supabase/supabase-js@2";
import webpush from "npm:web-push@3.6.7";

const cors = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type, x-cron-secret",
  "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
};
const admin = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!);
const json = (b: unknown, s = 200) => new Response(JSON.stringify(b), { status: s, headers: { ...cors, "Content-Type": "application/json" } });

async function keys() {
  const { data } = await admin.from("push_private").select("*").eq("id", 1).maybeSingle();
  let row = data;
  if (!row || !row.vapid_public) {
    const k = webpush.generateVAPIDKeys();
    const up = await admin.from("push_private")
      .upsert({ id: 1, vapid_public: k.publicKey, vapid_private: k.privateKey, cron_secret: row?.cron_secret ?? crypto.randomUUID() })
      .select().single();
    if (up.error) throw up.error;
    row = up.data;
  }
  webpush.setVapidDetails("https://jlghabitstracker.github.io", row.vapid_public, row.vapid_private);
  return row;
}

async function sendTo(userId: string, payload: Record<string, unknown>) {
  const { data: subs } = await admin.from("push_subs").select("endpoint,sub").eq("user_id", userId);
  let sent = 0; const errors: string[] = [];
  for (const s of subs ?? []) {
    try {
      await webpush.sendNotification(s.sub, JSON.stringify(payload), { TTL: 3600 });
      sent++;
    } catch (e) {
      const code = (e as { statusCode?: number }).statusCode;
      if (code === 404 || code === 410) await admin.from("push_subs").delete().eq("endpoint", s.endpoint);
      errors.push(String(code ?? (e as Error).message));
    }
  }
  return { sent, devices: (subs ?? []).length, errors };
}

function local(tz: string) {
  const p = new Intl.DateTimeFormat("en-CA", { timeZone: tz, year: "numeric", month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit", hour12: false }).formatToParts(new Date());
  const g = (t: string) => p.find((x) => x.type === t)!.value;
  const day = `${g("year")}-${g("month")}-${g("day")}`;
  const d = new Date(day + "T00:00:00Z");
  const back = (d.getUTCDay() + 6) % 7;
  const monday = new Date(d.getTime() - back * 864e5).toISOString().slice(0, 10);
  return { day, monday, mins: (Number(g("hour")) % 24) * 60 + Number(g("minute")) };
}
const mins = (t: string) => { const m = /^(\d{1,2}):(\d{2})$/.exec(t || ""); return m ? Number(m[1]) * 60 + Number(m[2]) : -1; };
const fmt = (n: number) => Number(n).toLocaleString("es-MX", { maximumFractionDigits: 2 });

type Habit = { id: string; name: string; type: string; goal?: number; unit?: string; days?: number[] };
function build(r: { habit?: string; pending?: boolean }, habits: Habit[], logs: Record<string, Record<string, number>>, today: string, quote = "", restDay = false) {
  const wd = (new Date(today + "T00:00:00Z").getUTCDay() + 6) % 7;
  const act = (h: Habit) => !restDay && (h.type === "week" || !Array.isArray(h.days) || !h.days.length || h.days.includes(wd));
  if (r.habit === "__frase") return quote ? { title: "Frase del día", body: quote, tag: "frase" } : null;
  const v = (h: Habit, k: string) => Number(logs[k]?.[h.id]) || 0;
  const weekCount = (h: Habit) => Object.keys(logs).filter((k) => v(h, k) >= 1).length;
  const done = (h: Habit) => h.type === "count" ? v(h, today) >= (h.goal || 0) && v(h, today) > 0
    : h.type === "week" ? (v(h, today) >= 1 || weekCount(h) >= (h.goal || 1)) : v(h, today) >= 1;
  const onlyPending = r.pending !== false;
  if (r.habit) {
    const h = habits.find((x) => x.id === r.habit);
    if (!h) return null;
    if (!act(h)) return null;
    if (onlyPending && done(h)) return null;
    const body = h.type === "count" ? `Llevas ${fmt(v(h, today))} de ${fmt(h.goal || 0)}${h.unit ? " " + h.unit : ""}.`
      : h.type === "week" ? `Llevas ${weekCount(h)} de ${h.goal} esta semana.`
      : done(h) ? "Ya lo marcaste hoy." : "Aún no lo marcas hoy.";
    return { title: h.name, body, tag: "h-" + h.id };
  }
  const todays = habits.filter(act);
  const left = todays.filter((h) => !done(h));
  if (!todays.length || (onlyPending && !left.length)) return null;
  const body = left.length ? `Te ${left.length === 1 ? "falta" : "faltan"} ${left.length}: ${left.map((h) => h.name).join(", ")}.` : "Ya cumpliste todo hoy.";
  return { title: "JLG Habits", body, tag: "resumen" };
}

// deno-lint-ignore no-explicit-any
function weekly(habits: any[], logs: Record<string, Record<string, number>>, monday: string, today: string, rest: Record<string, boolean>) {
  const days: string[] = [];
  for (let t = new Date(monday + "T00:00:00Z").getTime(); new Date(t).toISOString().slice(0, 10) <= today; t += 864e5) days.push(new Date(t).toISOString().slice(0, 10));
  // deno-lint-ignore no-explicit-any
  const v = (h: any, k: string) => Number(logs[k]?.[h.id]) || 0;
  // deno-lint-ignore no-explicit-any
  const done = (h: any, k: string) => h.type === "count" ? v(h, k) > 0 && v(h, k) >= (h.goal || 0) : v(h, k) >= 1;
  const rows = habits.map((h) => {
    if (h.type === "week") { const n = days.filter((k) => v(h, k) >= 1).length; return { n: h.name, ok: Math.min(n, h.goal || 1), of: h.goal || 1 }; }
    const act = days.filter((k) => !rest?.[k] && (!Array.isArray(h.days) || !h.days.length || h.days.includes((new Date(k + "T00:00:00Z").getUTCDay() + 6) % 7)));
    return { n: h.name, ok: act.filter((k) => done(h, k)).length, of: act.length };
  }).filter((r) => r.of > 0);
  if (!rows.length) return null;
  const tot = rows.reduce((a, r) => a + r.ok, 0), all = rows.reduce((a, r) => a + r.of, 0);
  const sorted = [...rows].sort((a, b) => b.ok / b.of - a.ok / a.of);
  const best = sorted[0], worst = sorted[sorted.length - 1];
  let body = `Cumpliste ${tot} de ${all} (${Math.round((tot / all) * 100)}%). Lo mejor: ${best.n} ${best.ok}/${best.of}.`;
  if (worst.n !== best.n && worst.ok < worst.of) body += ` A mejorar: ${worst.n} ${worst.ok}/${worst.of}.`;
  return { title: "Tu semana", body, tag: "semana" };
}

async function run() {
  const { data: cfgs, error } = await admin.from("user_config").select("user_id,habits,profile");
  if (error) throw error;
  let fired = 0;
  let quotes: string[] = [];
  try { quotes = await (await fetch("https://jlghabitstracker.github.io/frases.json")).json(); } catch { quotes = []; }
  if (!Array.isArray(quotes) || !quotes.length) quotes = ["La disciplina es hacerlo también el día que no tienes ganas."];
  let quotesEn: string[] = [];
  try { quotesEn = await (await fetch("https://jlghabitstracker.github.io/frases-en.json")).json(); } catch { quotesEn = []; }
  if (!Array.isArray(quotesEn) || !quotesEn.length) quotesEn = quotes;
  for (const c of cfgs ?? []) {
    const rems = c.profile?.reminders;
    if (!Array.isArray(rems) || !rems.length) continue;
    let now;
    try { now = local(c.profile.tz || "America/Mexico_City"); } catch { now = local("America/Mexico_City"); }
    const wdNow = (new Date(now.day + "T00:00:00Z").getUTCDay() + 6) % 7;
    const due = rems.filter((r: { time: string; dow?: number }) => { const m = mins(r.time); return m >= 0 && m <= now.mins && now.mins - m < 10 && (r.dow == null || r.dow === wdNow); });
    if (!due.length) continue;
    const { data: rows } = await admin.from("day_logs").select("day,v").eq("user_id", c.user_id).gte("day", now.monday).lte("day", now.day);
    const logs: Record<string, Record<string, number>> = {};
    for (const r of rows ?? []) logs[r.day] = r.v || {};
    for (const r of due) {
      const ins = await admin.from("push_sent").insert({ user_id: c.user_id, key: `${now.day}|${r.id}` });
      if (ins.error) continue;
      const msg = r.habit === "__semana" ? weekly(Array.isArray(c.habits) ? c.habits : [], logs, now.monday, now.day, c.profile?.rest || {}) : build(r, Array.isArray(c.habits) ? c.habits : [], logs, now.day, ((q: string[]) => q[Math.floor(new Date(now.day + "T00:00:00Z").getTime() / 864e5) % q.length])(c.profile?.quoteLang === "en" ? quotesEn : quotes), !!c.profile?.rest?.[now.day]);
      if (!msg) continue;
      if (r.habit === "__frase" && c.profile?.quoteLang === "en") msg.title = "Quote of the day";
      await sendTo(c.user_id, msg);
      fired++;
    }
  }
  await admin.from("push_sent").delete().lt("created_at", new Date(Date.now() - 3 * 864e5).toISOString());
  return { ok: true, fired };
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: cors });
  try {
    const row = await keys();
    if (req.method === "GET") return json({ publicKey: row.vapid_public });
    const body = await req.json().catch(() => ({}));
    if (req.headers.get("x-cron-secret") === row.cron_secret) return json(await run());
    const token = (req.headers.get("Authorization") || "").replace(/^Bearer /, "");
    const { data } = await admin.auth.getUser(token);
    if (!data?.user) return json({ error: "no autorizado" }, 401);
    if (body.action === "test") return json(await sendTo(data.user.id, { title: "JLG Habits", body: "Las notificaciones ya funcionan.", tag: "prueba" }));
    return json({ error: "acción desconocida" }, 400);
  } catch (e) {
    return json({ error: String((e as Error).message ?? e) }, 500);
  }
});
